func esc(_ s: String) -> String {
    var out = ""
    for c in s.unicodeScalars {
        switch c {
        case "\"": out += "\\\""
        case "\\": out += "\\\\"
        case "\n": out += "\\n"
        case "\t": out += "\\t"
        default:
            if c.value < 0x20 {
                let hex = String(c.value, radix: 16)
                out += "\\u" + String(repeating: "0", count: 4 - hex.count) + hex
            } else { out.unicodeScalars.append(c) }
        }
    }
    return out
}

nonisolated(unsafe) var parts: [String] = []
func emit(_ key: String, _ source: String, _ render: String) {
    parts.append("\"\(esc(key))\":{\"source\":\"\(esc(source))\",\"render\":\"\(esc(render))\"}")
}
/// A render that is a value rather than a diagnostic — the happy path.
func emitValue(_ key: String, _ source: String, _ value: Any) {
    emit(key, source, String(describing: value))
}
/// The same, for a call that can throw. An example that stops working should say so in
/// the output rather than killing the run: with thirty of them, one `try!` trap means one
/// rebuild per problem.
func emitTry(_ key: String, _ source: String, _ body: () throws -> Any) {
    do { emit(key, source, String(describing: try body())) }
    catch { emit(key, source, "EXAMPLE FAILED: \(error)") }
}

// The value model, and the four date-time kinds.
let tomlValueModel = """
odt = 1979-05-27T07:32:00Z
ldt = 1979-05-27T07:32:00
ld  = 1979-05-27
lt  = 07:32:00
hex = 0xDEAD_BEEF
"""
let td = try! TOML.parse(tomlValueModel)
func kind(_ k: String) -> String {
    guard let dt = td[k]?.dateTime else { return "nil" }
    switch dt {
    case .offsetDateTime(let s): return "offsetDateTime(\(s))"
    case .localDateTime(let s): return "localDateTime(\(s))"
    case .localDate(let s): return "localDate(\(s))"
    case .localTime(let s): return "localTime(\(s))"
    }
}
emit("toml.valueModel", tomlValueModel,
     "t[\"odt\"]?.dateTime → \(kind("odt"))\n"
     + "t[\"ldt\"]?.dateTime → \(kind("ldt"))\n"
     + "t[\"ld\"]?.dateTime  → \(kind("ld"))\n"
     + "t[\"lt\"]?.dateTime  → \(kind("lt"))\n"
     + "t[\"hex\"]?.int      → \(td["hex"]?.int.map(String.init) ?? "nil")")
